\cohead{\Large\textbf{Vektorbegriff}}
\fakesubsection{Vektorbegriff}
\begin{tcolorbox}[height=4cm]
	\textbf{Vektoren:}
	
	\bigskip
	
	\textcolor{loestc}{Ein Vektor ist letztendlich ein Pfeil. Die Menge aller Pfeile bzw. Vektoren, die die gleiche Richtung und Länge haben, bezeichnet man auch als Pfeilklasse.}
	
	\bigskip
	
	\textcolor{loestc}{Vektoren schreibt man mit einem kleinen Pfeil oberhalb der Größe, z.B. \(\vv{a}\).}
\end{tcolorbox}

			\textcolor{loes}{Die beiden Vektoren \(\vv{AB}\) und \(\vv{CD}\) sind gleich lang und zeigen in die gleiche Richtung, daher stellen sie denselben Vektor dar: \(\vv{AB}=\vv{CD}\)}

			Zeichne 2 weitere Punkte ein, deren Verbindungsvektor ebenfalls \(\vv{AB}\) ist.

\medskip

Wir haben oben bereits den Begriff Verbindungsvektor intuitiv eingeführt, den wir jetzt, neben anderen, noch genauer bestimmen wollen:
\begin{tcolorbox}[sidebyside, height=5cm]
	\textbf{Verbindungsvektoren}
	
	\textcolor{loestc}{Ein Vektor, der zwei Punkte verbindet.}
	
	\textcolor{loestc}{Schreibweise: Verbindungsvektor vom Punkt \(A\) zum Punkt \(B\): \(\vv{AB}\)}
	\tcblower
	\textbf{Ortsvektoren}
	
	\textcolor{loestc}{Der Ortsvektor eines Punktes \(A\) ist der Vektor, der vom Ursprung \(O\) auf den Punkt \(A\) zeigt:}
	
	\textcolor{loestc}{Ortsvektor \(A\): \(\vv{OA}\).}
\end{tcolorbox}

\begin{tcolorbox}[sidebyside, height=5cm]
	\textbf{Nullvektor}
	
	\textcolor{loestc}{Der Vektor mit der Länge Null wird als Nullvektor bezeichnet:}
	
	\textcolor{loestc}{Schreibweise: \(\vv{o}\)}
	\tcblower
	\textbf{Gegenvektor}
	
	\textcolor{loestc}{Der Gegenvektor eines Vektors hat die gleiche Länge und die genau entgegengesetzte Richtung:}
	
	\textcolor{loestc}{Gegenvektor zu \(\vv{AB}\): \(\vv{BA}=-\vv{AB}\).}
\end{tcolorbox}
